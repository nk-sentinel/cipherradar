Quick Install
# Download and extract
curl -sL https://github.com/nk-sentinel/cipherradar/releases/latest/download/cbom_0.1.0_darwin_arm64.tar.gz | tar xz
# Move to PATH
sudo mv cbom /usr/local/bin/
# Verify
cbom version
# Install OpenGrep (optional — for Pass 2 taint analysis)
cbom install-tools
Connect CLI to Dashboard
# Generate an API key in Settings > API Keys, then:
export CBOM_API_KEY="your-api-key-here"
export CBOM_API_URL="https://cipherradar.yourcompany.com/api/v1"
# Scan and upload results to dashboard
cbom scan . --format cyclonedx-json --output cbom.json
curl -X POST $CBOM_API_URL/scans \
  -H "Authorization: Bearer $CBOM_API_KEY" \
  -F "cbom=@cbom.json"
